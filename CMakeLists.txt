cmake_minimum_required(VERSION 3.21)
project(CA_MAME VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
FetchContent_Declare(zlib URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
# Prefer system zlib when available; otherwise fetch the portable source.
find_package(ZLIB QUIET)
if(NOT ZLIB_FOUND)
    FetchContent_MakeAvailable(zlib)
    add_library(ZLIB::ZLIB ALIAS zlibstatic)
endif()
add_library(ca_mame_core
    src/core/Machine.cpp
    src/core/MachineRegistry.cpp
    src/rom/RomManager.cpp
    src/bus/MemoryBus.cpp
    src/cpu/Intel8080.cpp
    src/cpu/PowerPC602.cpp
    src/drivers/SpaceInvaders.cpp
)
target_link_libraries(ca_mame_core PUBLIC ZLIB::ZLIB)
target_include_directories(ca_mame_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(ca_mame_core PRIVATE /W4)
else()
    target_compile_options(ca_mame_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(ca-mame src/frontend/main.cpp)
target_link_libraries(ca-mame PRIVATE ca_mame_core)

option(CA_MAME_ENABLE_SDL3 "Build the optional SDL3 arcade frontend" OFF)
if(CA_MAME_ENABLE_SDL3)
    find_package(SDL3 CONFIG REQUIRED)
    add_executable(ca-mame-gun-test src/frontend/light_gun_test.cpp)
    target_link_libraries(ca-mame-gun-test PRIVATE ca_mame_core SDL3::SDL3)
    add_executable(ca-mame-sdl src/frontend/sdl_main.cpp)
    target_link_libraries(ca-mame-sdl PRIVATE ca_mame_core SDL3::SDL3)
endif()
